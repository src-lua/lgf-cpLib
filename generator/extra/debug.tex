\section{Debugging}

\vspace{-10pt}

{\fontsize{7}{9}\selectfont
\begin{verbatim}
-g++ -O2 -std=c++17 -Wall -Wextra
    -fsanitize=address,undefined
    -fno-sanitize-recover=all -g
\end{verbatim}
}

\vspace{-8pt}

\textbf{Antes de submeter:}
{\fontsize{6.5}{8}\selectfont
\begin{itemize}[noitemsep,topsep=2pt,leftmargin=*]
\item Escreva casos simples se o sample não for suficiente.
\item Limite de tempo apertado? Gere casos máximos.
\item Memória está ok?
\item Algo pode dar overflow?
\item Está submetendo o arquivo certo?
\end{itemize}
}

\textbf{Wrong Answer:}
{\fontsize{6.5}{8}\selectfont
\begin{itemize}[noitemsep,topsep=2pt,leftmargin=*]
\item Printe sua solução e outputs de debug.
\item Está limpando estruturas entre casos de teste?
\item O algoritmo cobre todo o range de entrada?
\item Releia o enunciado completo.
\item Todos os casos de borda estão corretos?
\item Entendeu o problema corretamente?
\item Variáveis não inicializadas?
\item Overflow?
\item Confundindo N/M, i/j, etc.?
\item Tem certeza que o algoritmo funciona?
\item Que casos especiais você não considerou?
\item As funções da STL funcionam como você acha?
\item Adicione \texttt{assert}s e ressubmeta.
\item Rode em casos criados manualmente.
\item Execute o algoritmo num caso simples à mão.
\item Explique o algoritmo para um colega.
\item Peça para o colega revisar seu código.
\item Dê uma volta, vá ao banheiro.
\item O formato de saída está correto? (espaços, newlines)
\item Reescreva do zero ou peça ao colega.
\end{itemize}
}

\textbf{Runtime Error:}
{\fontsize{6.5}{8}\selectfont
\begin{itemize}[noitemsep,topsep=2pt,leftmargin=*]
\item Testou todos os casos de borda localmente?
\item Variáveis não inicializadas?
\item Lendo/escrevendo fora do range de algum vetor?
\item Algum \texttt{assert} que pode falhar?
\item Divisão por zero? (\texttt{mod 0}?)
\item Recursão infinita possível?
\item Ponteiros ou iteradores invalidados?
\item Usando memória demais?
\end{itemize}
}

\textbf{Time Limit Exceeded:}
{\fontsize{6.5}{8}\selectfont
\begin{itemize}[noitemsep,topsep=2pt,leftmargin=*]
\item Tem loop infinito?
\item Qual é a complexidade do algoritmo?
\item Está copiando dados desnecessariamente? (use referências)
\item Tamanho da entrada/saída? (considere \texttt{scanf})
\item Evite \texttt{map}/\texttt{vector}; use arrays/\texttt{unordered\_map}.
\item O que seus colegas acham do algoritmo?
\end{itemize}
}

\textbf{Memory Limit Exceeded:}
{\fontsize{6.5}{8}\selectfont
\begin{itemize}[noitemsep,topsep=2pt,leftmargin=*]
\item Quanto de memória seu algoritmo realmente precisa?
\item Está limpando estruturas entre casos de teste?
\end{itemize}
}
